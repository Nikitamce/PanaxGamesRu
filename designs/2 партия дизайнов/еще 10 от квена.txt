/* ================= POINTER FX ================= */
const canHover=matchMedia('(hover:hover) and (pointer:fine)').matches;
function bindFX(){
  if(!canHover)return;
  $$('[data-fx]').forEach(el=>{
    const kind=el.dataset.fx;
    el.addEventListener('pointerenter',()=>el.classList.remove('rest'));
    el.addEventListener('pointermove',ev=>{
      const r=el.getBoundingClientRect(),px=(ev.clientX-r.left)/r.width,py=(ev.clientY-r.top)/r.height;
      if(kind==='tilt'){el.style.setProperty('--rx',((py-.5)*-7).toFixed(2)+'deg');el.style.setProperty('--ry',((px-.5)*9).toFixed(2)+'deg');el.style.setProperty('--mx',(px*100).toFixed(1)+'%');el.style.setProperty('--my',(py*100).toFixed(1)+'%')}
      else if(kind==='holo'){el.style.setProperty('--rx',((py-.5)*-16).toFixed(2)+'deg');el.style.setProperty('--ry',((px-.5)*18).toFixed(2)+'deg');el.style.setProperty('--mx',(px*100).toFixed(1)+'%');el.style.setProperty('--my',(py*100).toFixed(1)+'%');el.style.setProperty('--mxn',px.toFixed(3));el.style.setProperty('--myn',py.toFixed(3));el.style.setProperty('--fo','1')}
      else if(kind==='pan'){el.style.setProperty('--px',((px-.5)*-14).toFixed(1)+'px');el.style.setProperty('--py',((py-.5)*-10).toFixed(1)+'px')}
    });
    el.addEventListener('pointerleave',()=>{
      el.classList.add('rest');
      ['--rx','--ry'].forEach(p=>el.style.setProperty(p,'0deg'));
      if(kind==='holo')el.style.setProperty('--fo','0');
      if(kind==='pan'){el.style.setProperty('--px','0px');el.style.setProperty('--py','0px')}
    });
  });
}
